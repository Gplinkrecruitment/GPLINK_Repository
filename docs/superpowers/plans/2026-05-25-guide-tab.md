> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a "Guide" tab to the RSO admin with folder-organized Scribe tutorial embeds, manageable by CEO only.

**Architecture:** New Supabase tables (`guide_folders`, `guide_items`) with CRUD API endpoints in server.js. Guide tab UI added to admin.html following existing tab/panel patterns. Folder list on left, slide-out Scribe iframe panel on right.

**Tech Stack:** Supabase (PostgreSQL), Node.js (server.js), vanilla JS/HTML (admin.html)

```sql
CREATE TABLE IF NOT EXISTS guide_folders (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  sort_order INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

ALTER TABLE guide_folders ENABLE ROW LEVEL SECURITY;

CREATE POLICY "guide_folders_service_role" ON guide_folders
  FOR ALL USING (auth.role() = 'service_role');

CREATE TABLE IF NOT EXISTS guide_items (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  folder_id UUID NOT NULL REFERENCES guide_folders(id) ON DELETE CASCADE,
  title TEXT NOT NULL,
  scribe_url TEXT NOT NULL,
  sort_order INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_guide_items_folder ON guide_items(folder_id);

ALTER TABLE guide_items ENABLE ROW LEVEL SECURITY;

CREATE POLICY "guide_items_service_role" ON guide_items

INSERT INTO guide_folders (name, sort_order) VALUES
```

```javascript
if (req.method === 'GET' && pathname === '/api/admin/guide/folders') {

  if (!adminCtx) return;
  try {
    var foldersRes = await supabaseDbRequest('guide_folders', 'select=*&order=sort_order.asc,created_at.asc');
    if (!foldersRes.ok) { sendJson(res, 500, { ok: false, message: 'Failed to load folders.' }); return; }
    var itemsRes = await supabaseDbRequest('guide_items', 'select=*&order=sort_order.asc,created_at.asc');
    var items = (itemsRes.ok && Array.isArray(itemsRes.data)) ? itemsRes.data : [];
    var folders = (Array.isArray(foldersRes.data) ? foldersRes.data : []).map(function(f) {
      return Object.assign({}, f, { items: items.filter(function(i) { return i.folder_id === f.id; }) });
    });
    sendJson(res, 200, { ok: true, folders: folders });
  } catch (err) {
    sendJson(res, 500, { ok: false, message: err.message });
  }
  return;
}
```

```javascript
// ── Guide Tab ──────────────────────────────────────────────
async function loadGuideData(){
  S.guide.loading=true;
  try{
    var r=await fetch('/api/admin/guide/folders',{credentials:'include'});
    var d=await r.json();
    if(d.ok)S.guide.folders=d.folders||[];
  }catch(e){console.error('[Guide] load error',e);}
  S.guide.loading=false;
}

function renderGuidePanel(){
  var el=document.getElementById('guidePanel');if(!el)return;
  var folders=S.guide.folders||[];
  var sa=isSA();
  var folderListHtml=folders.map(function(f){
    var expanded=S.guide.expandedFolderId===f.id;
    var items=f.items||[];
    var itemsHtml='';
    if(expanded){
      itemsHtml='<div class="guide-folder-items">'+
        items.map(function(item){
          var active=S.guide.selectedItemId===item.id;
          return '<div class="guide-item'+(active?' active':'')+'" data-guide-item="'+item.id+'" data-scribe-url="'+item.scribe_url.replace(/"/g,'&quot;')+'" data-item-title="'+item.title.replace(/"/g,'&quot;')+'">'+
            item.title+
            (sa?'<button class="guide-mgmt-btn" data-edit-item="'+item.id+'" title="Edit">&#9998;</button><button class="guide-mgmt-btn" data-delete-item="'+item.id+'" title="Delete">&times;</button>':'')+
          '</div>';
        }).join('')+
        (items.length===0?'<div style="padding:8px 12px;font-size:11px;color:var(--muted);">No guides in this folder yet</div>':'')+
        (sa?'<button class="guide-add-btn" data-add-item-folder="'+f.id+'">+ Add Guide</button>':'')+
      '</div>';
    }
    return '<div class="guide-folder">'+
      '<div class="guide-folder-header'+(expanded?' active':'')+'" data-guide-folder="'+f.id+'">'+
        '<span>'+(expanded?'&#9660;':'&#9654;')+'</span> &#128193; '+f.name+
        '<span class="guide-folder-count">'+items.length+'</span>'+
        (sa?'<button class="guide-mgmt-btn" data-edit-folder="'+f.id+'" title="Rename">&#9998;</button><button class="guide-mgmt-btn" data-delete-folder="'+f.id+'" title="Delete">&times;</button>':'')+
      '</div>'+itemsHtml+
    '</div>';
  }).join('');

  var selectedItem=null;
  if(S.guide.selectedItemId){
    folders.forEach(function(f){(f.items||[]).forEach(function(i){if(i.id===S.guide.selectedItemId)selectedItem=i;});});
  }

  var detailHtml='';
  if(selectedItem){
    detailHtml='<div class="guide-detail-header"><h3>'+selectedItem.title+'</h3><button onclick="S.guide.selectedItemId=null;renderGuidePanel();" style="border:none;background:transparent;cursor:pointer;font-size:18px;color:var(--muted);">&times;</button></div>'+
      '<div class="guide-detail-body"><iframe src="'+selectedItem.scribe_url+'" allowfullscreen></iframe></div>';
  }else{
    detailHtml='<div class="guide-empty"><div><div style="font-size:40px;margin-bottom:8px;">&#127891;</div>Select a guide to view the tutorial</div></div>';
  }

  el.innerHTML='<div class="guide-wrap">'+
    '<div class="guide-folders">'+
      '<div style="display:flex;align-items:center;justify-content:space-between;margin-bottom:12px;"><h3 style="margin:0;font-size:15px;font-weight:700;">Guides</h3>'+
      (sa?'<button class="guide-add-btn" data-add-folder>+ Add Folder</button>':'')+
      '</div>'+
      (folders.length===0?'<div class="guide-empty" style="height:auto;padding:20px;"><div>No guides yet</div></div>':folderListHtml)+
    '</div>'+
    '<div class="guide-detail">'+detailHtml+'</div>'+
  '</div>';
}
```

```javascript
document.addEventListener('click',function(e){
  // Guide folder toggle
  var fh=e.target.closest('[data-guide-folder]');
  if(fh&&!e.target.closest('.guide-mgmt-btn')){
    var fid=fh.getAttribute('data-guide-folder');
    S.guide.expandedFolderId=S.guide.expandedFolderId===fid?null:fid;
    renderGuidePanel();return;
  }
  // Guide item select
  var gi=e.target.closest('[data-guide-item]');
  if(gi&&!e.target.closest('.guide-mgmt-btn')){
    S.guide.selectedItemId=gi.getAttribute('data-guide-item');
    renderGuidePanel();return;
  }
  // Add folder (CEO)
  if(e.target.closest('[data-add-folder]')){
    var name=prompt('Folder name:');
    if(!name)return;
    fetch('/api/admin/guide/folders',{method:'POST',credentials:'include',headers:{'Content-Type':'application/json'},body:JSON.stringify({name:name})})
      .then(function(r){return r.json();}).then(function(d){if(d.ok){S.guide.folders=null;loadGuideData().then(renderGuidePanel);}else{alert(d.message||'Failed');}});
    return;
  }
  // Add item (CEO)
  var addItemBtn=e.target.closest('[data-add-item-folder]');
  if(addItemBtn){
    var folderId=addItemBtn.getAttribute('data-add-item-folder');
    var title=prompt('Guide title:');if(!title)return;
    var url=prompt('Scribe embed URL:');if(!url)return;
    fetch('/api/admin/guide/items',{method:'POST',credentials:'include',headers:{'Content-Type':'application/json'},body:JSON.stringify({folder_id:folderId,title:title,scribe_url:url})})
      .then(function(r){return r.json();}).then(function(d){if(d.ok){S.guide.folders=null;loadGuideData().then(renderGuidePanel);}else{alert(d.message||'Failed');}});
    return;
  }
  // Edit folder (CEO)
  var editFolderBtn=e.target.closest('[data-edit-folder]');
  if(editFolderBtn){
    var fid=editFolderBtn.getAttribute('data-edit-folder');
    var current=(S.guide.folders||[]).find(function(f){return f.id===fid;});

    fetch('/api/admin/guide/folders/'+encodeURIComponent(fid),{method:'PUT',credentials:'include',headers:{'Content-Type':'application/json'},body:JSON.stringify({name:newName})})
      .then(function(r){return r.json();}).then(function(d){if(d.ok){S.guide.folders=null;loadGuideData().then(renderGuidePanel);}else{alert(d.message||'Failed');}});
    return;
  }
  // Delete folder (CEO)
  var delFolderBtn=e.target.closest('[data-delete-folder]');
  if(delFolderBtn){
    var fid=delFolderBtn.getAttribute('data-delete-folder');
    if(!confirm('Delete this folder and all its guides?'))return;
    fetch('/api/admin/guide/folders/'+encodeURIComponent(fid),{method:'DELETE',credentials:'include'})
      .then(function(r){return r.json();}).then(function(d){if(d.ok){S.guide.folders=null;S.guide.selectedItemId=null;loadGuideData().then(renderGuidePanel);}else{alert(d.message||'Failed');}});
    return;
  }
  // Edit item (CEO)
  var editItemBtn=e.target.closest('[data-edit-item]');
  if(editItemBtn){
    var iid=editItemBtn.getAttribute('data-edit-item');
    var currentItem=null;(S.guide.folders||[]).forEach(function(f){(f.items||[]).forEach(function(i){if(i.id===iid)currentItem=i;});});
```
